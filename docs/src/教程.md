### 1.2.2 不会写建表语句？三条路（评分只看表能用，不看你写没写 SQL）

**路线一：SSMS 图形界面建表（不用写一行 SQL，推荐先练熟这个）**

1. 对象资源管理器 → 展开你的数据库（MyDb）→ 右键"**表**" → 新建 → **表**。
2. 在表格设计器里一行行填（这就是"可视化建表"）：

| 列名 | 数据类型 | 允许 Null | 备注 |
|---|---|---|---|
| userid | bigint | 取消勾选 | 右键此行 → **设置主键**（钥匙图标）；下方"列属性"→ **标识规范** → "(是标识)" 改为 **是**（= SQL 里的 IDENTITY 自增） |
| username | nvarchar(32) | 取消勾选 | 中文必须 nvarchar；右键 → 索引/键 → 添加 → 类型选"唯一键"，列选 username（注册判重兜底） |
| password | nvarchar(64) | 取消勾选 | |
| phone | nvarchar(11) | 取消勾选 | |

3. `Ctrl+S` → 输入表名 `user` → 确定。其余表（service/orders）照抄，字段类型规则：**文字用 nvarchar(长度)、金额用 decimal(10,2)、日期用 datetime、ID 列用 int/bigint 自增**。
4. 造测试数据（不用 INSERT）：右键表 → **"编辑前 200 行"** → 直接在格子填 → 关窗口自动保存。

**路线二：万能模板填空法（把建表当"改名字"）**

背一个最小模板，任何表都套它：

```sql
CREATE TABLE 表名 (
    id       INT IDENTITY(1,1) PRIMARY KEY,   -- 主键，自动编号
    字段1    NVARCHAR(50) NOT NULL,           -- 文本：长度看内容
    金额字段  DECIMAL(10,2) NOT NULL,          -- 价格类
    日期字段  DATETIME NOT NULL               -- 日期类
);
```

**路线三：让程序自己建表（最强兜底，双击 exe 全自动）**

把建表脚本写进代码，程序一启动自动"建库→建表→造数据"，你一个字都不用手写。放一个独立类 `DbInit.cs`，在 `Program.Main` 里最先调用：

```csharp
using System.Data.SqlClient;

public class DbInit
{
    public static void Init()
    {
        // 1. 先连 master（因为 MyDb 还没建，直接连会报错）
        using (var c = new SqlConnection("Data Source=localhost;Initial Catalog=master;Integrated Security=True"))
        {
            c.Open(); Run(c, "IF DB_ID('MyDb') IS NULL CREATE DATABASE MyDb;");
        }
        // 2. 再连 MyDb 建表 + 造数据
        using (var c = new SqlConnection("Data Source=localhost;Initial Catalog=MyDb;Integrated Security=True"))
        {
            c.Open();
            Run(c, @"IF OBJECT_ID('[user]') IS NULL CREATE TABLE [user] (
                        userid BIGINT IDENTITY(1,1) PRIMARY KEY,
                        username NVARCHAR(32) NOT NULL UNIQUE,
                        password NVARCHAR(64) NOT NULL,
                        phone NVARCHAR(11) NOT NULL DEFAULT '')");
            Run(c, "IF OBJECT_ID('service') IS NULL CREATE TABLE service ( ...同上或接赛题字段... )");
            Run(c, "IF OBJECT_ID('orders') IS NULL CREATE TABLE orders ( ... )");
            Run(c, "IF NOT EXISTS (SELECT 1 FROM [user] WHERE username='admin') " +
                   "INSERT INTO [user] (username,password,phone) VALUES ('admin','admin123','13800000000')");
        }
    }
    static void Run(SqlConnection c, string sql)
    {
        using (var cmd = c.CreateCommand()) { cmd.CommandText = sql; cmd.ExecuteNonQuery(); }
    }
}

// Program.cs 主入口：先建库表数据，再启动登录页
static void Main()
{
    DbInit.Init();                              // ← 加这一行
    Application.EnableVisualStyles();
    Application.Run(new LoginForm());
}
```

> 效果：无论你在哪台机器上跑，只要 SQL Server 在跑，程序一开全自动就绪。**建表语句还是同一份 SQL，但它"住在代码里"，你不需要现场背、不会手抖。**

**三条路怎么用（定位分工）**：路线三做**保底**（赛前写好，现场双击 exe 自动建好库表数据）；路线一应付**突发**（现场临时要加表、加字段，点鼠标 2 分钟搞定）；路线二用来**理解字段含义**（看懂别人给你的表结构，或对着赛题字段自己套）。现在就用 SSMS 图形界面把"右键建表 → 填字段 → 设主键/标识 → 编辑前 200 行"练三遍，点成肌肉记忆，现场就不慌了。

**② DAL 数据库大门（全项目共用，默写）**：

```csharp
using System;
using System.Collections.Generic;
using System.Data;
using System.Data.SqlClient;

namespace DAL
{
    public class DBHelper
    {
        private static readonly DBHelper _instance = new DBHelper();
        public static DBHelper Instance => _instance;          // 用法：DBHelper.Instance
        private SqlConnection conn;

        private DBHelper()
        {
            const string connStr = "Data Source=localhost;Initial Catalog=MyDb;Integrated Security=True";
            conn = new SqlConnection(connStr);
        }

        // 唯一查询入口：SQL + 参数字典 → DataTable。参数化防注入，禁止字符串拼接用户输入。
        public DataTable ExecuteQuery(string sql, Dictionary<string, object> param = null)
        {
            using (SqlDataAdapter da = new SqlDataAdapter(sql, conn))
            {
                if (param != null)
                    foreach (var kv in param)
                        da.SelectCommand.Parameters.AddWithValue(kv.Key, kv.Value);
                DataTable dt = new DataTable();
                da.Fill(dt);
                return dt;
            }
        }
    }
}
```

**③ DAL 用户数据方法（注册查重/插入/登录查询）**：

```csharp
using System.Collections.Generic;
using System.Data;
using Module;

namespace DAL
{
    public class UserDAL
    {
        // 按用户名查（判重用 + 登录用）——没有返回 null
        public static UserInfo GetByName(string name)
        {
            DataTable dt = DBHelper.Instance.ExecuteQuery(
                "SELECT userid, username, password, phone FROM [user] WHERE username=@u",
                new Dictionary<string, object> { { "@u", name } });
            if (dt.Rows.Count == 0) return null;
            UserInfo u = new UserInfo
            {
                UserId = (long)dt.Rows[0]["userid"],
                UserName = dt.Rows[0]["username"].ToString(),
                UserPassword = dt.Rows[0]["password"].ToString(),
                UserPhone = dt.Rows[0]["phone"].ToString()
            };
            return u;
        }

        // 注册 = 插入
        public static void Insert(string name, string pwd, string phone)
        {
            DBHelper.Instance.ExecuteQuery(
                "INSERT INTO [user] (username, password, phone) VALUES (@u, @p, @t)",
                new Dictionary<string, object> { { "@u", name }, { "@p", pwd }, { "@t", phone } });
        }
    }
}
```

**④ BLL 业务规则（校验与判断放这里，UI 不碰 SQL）**：

```csharp
using System.Text.RegularExpressions;
using Module;

namespace BLL
{
    public class UserBLL
    {
        // 登录：用户存在 且 密码一致（两条件缺一不可）
        public static bool Login(string name, string pwd)
        {
            UserInfo u = DAL.UserDAL.GetByName(name);
            return u != null && u.UserPassword == pwd;
        }

        // 注册：返回 null=成功；返回字符串=错误提示（界面直接显示）
        public static string Register(string name, string pwd, string pwd2, string phone)
        {
            if (string.IsNullOrEmpty(name))            return "用户名不能为空";
            if (DAL.UserDAL.GetByName(name) != null)   return "用户名已存在";                       // 唯一性
            if (!Regex.IsMatch(pwd, @"^(?=.*[A-Za-z])(?=.*\d).{8,}$"))  return "密码至少8位，且必须包含字母和数字";
            if (pwd != pwd2)                           return "两次密码不一致";
            if (!Regex.IsMatch(phone, @"^\d{11}$"))    return "联系电话必须为11位数字";
            DAL.UserDAL.Insert(name, pwd, phone);
            return null;
        }
    }
}
```

### 3.3.1 验证码模拟器（内存版，整段拷贝）

```java
// SmsCodeManager.java —— 模拟短信验证码：6位、1分钟过期、可重发
public class SmsCodeManager {
    private static final java.util.Map<String, CodeInfo> CODES = new java.util.HashMap<>();

    public static String generate(String phone) {
        String code = String.valueOf(new java.util.Random().nextInt(900000) + 100000); // 6位
        CODES.put(phone, new CodeInfo(code, System.currentTimeMillis() + 60_000));     // 1分钟
        return code;
    }

    /** true=验证通过（码对且未过期） */
    public static boolean verify(String phone, String code) {
        CodeInfo info = CODES.get(phone);
        return info != null && System.currentTimeMillis() <= info.expireAt && info.code.equals(code);
    }

    static class CodeInfo {
        String code; long expireAt;
        CodeInfo(String c, long e) { code = c; expireAt = e; }
    }
}
```

## 3.4 数据来源：Modul1 接口（HttpURLConnection + org.json，零依赖）

样题原文："本模块数据来源 Modul1 接口。"——手机端从接口拉数据，别用假数据。接口地址开赛时会给出；下面是**通用拉取模板**（GET 列表 JSON）：

```java
// ApiHelper.java
public class ApiHelper {
    public static java.util.List<java.util.HashMap<String, Object>> getServices(String url)
            throws Exception {
        // 1. 开连接（模拟器访问电脑本机用 10.0.2.2，真机用电脑局域网 IP）
        HttpURLConnection c = (HttpURLConnection) new java.net.URL(url).openConnection();
        c.setConnectTimeout(5000);
        c.setReadTimeout(5000);
        c.setRequestMethod("GET");

        // 2. 读响应全文
        java.io.InputStream is = c.getInputStream();
        java.io.BufferedReader r = new java.io.BufferedReader(new java.io.InputStreamReader(is, "UTF-8"));
        StringBuilder sb = new StringBuilder();
        String line;
        while ((line = r.readLine()) != null) sb.append(line);
        is.close();

        // 3. 解析 JSON 数组 → 每个元素是 HashMap（键名 = 接口字段名）
        //    如果接口包一层 { "data": [...] }，就先取 jsonObj.getJSONArray("data")
        org.json.JSONArray arr = new org.json.JSONArray(sb.toString());
        java.util.List<java.util.HashMap<String, Object>> list = new java.util.ArrayList<>();
        for (int i = 0; i < arr.length(); i++) {
            org.json.JSONObject o = arr.getJSONObject(i);
            java.util.HashMap<String, Object> m = new java.util.HashMap<>();
            java.util.Iterator<String> keys = o.keys();
            while (keys.hasNext()) {
                String k = keys.next();
                m.put(k, o.isNull(k) ? "" : o.getString(k));
            }
            list.add(m);
        }
        return list;
    }
}

// Activity 里调用（网络必须异步，取完回主线程刷新）
private void loadData() {
    new Thread(() -> {
        try {
            java.util.List<java.util.HashMap<String, Object>> data =
                    ApiHelper.getServices("http://10.0.2.2:8080/api/service");
            runOnUiThread(() -> { adapter = new ServiceAdapter(this, data); lv.setAdapter(adapter); });
        } catch (Exception e) {
            runOnUiThread(() -> Toast.makeText(this, "加载失败：" + e.getMessage(), Toast.LENGTH_SHORT).show());
        }
    }).start();
}
```

三个必背坑：**模拟器访问电脑本机用 `10.0.2.2`**；**Android 9+ 明文 http 需 `usesCleartextTraffic="true"`**；**网络不能卡主线程**（`Thread`+`runOnUiThread` 即可，别学 RxJava）。

    @Override public long getItemId(int i) { return i; }

        ((android.widget.TextView) convertView.findViewById(R.id.tv_price)).setText(
                String.valueOf(m.get("price")) + "元");

    }
}

# 附录B 赛场常见错误与急救

| 症状 | 原因 | 急救 |
|---|---|---|
| 连不上数据库 | 库名/服务器名不一致 | 检查连接串 `Database=MyDb`；SSMS 里确认库存在 |
| 中文乱码 | 缺 NVARCHAR / 编码问题 | 表字段用 `NVARCHAR`；连接串加 `;Character Set=utf8`（MySQL 时） |
| Android 接口 500/404 | 接口地址不对 | 模拟器用 `10.0.2.2` 替代 `localhost`；真机用电脑局域网 IP |
| 明文 http 被拦 | Android 9+ 默认禁止 | Manifest 加 `android:usesCleartextTraffic="true"` |
| 列表卡/闪退 | 主线程做网络 | 网络全部放 `Thread`，切 UI 用 `runOnUiThread` |
| 界面没有滚动条 | 控件尺寸撑满 | 外层套 `ScrollView`/`ListView`；WinForms 检查 `ScrollBars` 属性 |
| 打包失败缺依赖 | 没网下载库 | 只用 SDK 内置 API；确认 Gradle 缓存 |
| 忘记 keystore 密码 | 签名 APK 打不出 | 密码写进备忘录文本放 Modul2 文件夹一起提交 |

> 最后一句话：**分数 = 明牌题做满分（登录/注册） + 硬性约定不丢分（日期/滚动条/图片/提示） + 每页三件套齐全 + 按时交对文件名的压缩包。** 祝你赛场顺利。
